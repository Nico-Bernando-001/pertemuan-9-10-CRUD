<?php

function ambilSemuaMahasiswa($conn) {

    $sql =
        "SELECT
            mahasiswa.id,
            mahasiswa.nim,
            mahasiswa.nama,
            mahasiswa.email,
            program_studi.nama_prodi
        FROM mahasiswa
        JOIN program_studi
            ON mahasiswa.id_prodi =
                program_studi.id_prodi
        ORDER BY mahasiswa.nama ASC";
    
    return $conn->query($sql);
}


function ambilSemuaProdi($conn) {

    $sql =
        "SELECT *
        FROM program_studi
        ORDER BY nama_prodi ASC";
    
    return $conn->query($sql);
}


function tambahMahasiswa(
    $conn,
    $nim,
    $nama,
    $email,
    $password,
    $idProdi
) {

    $stmt = $conn->prepare(
        "INSERT INTO mahasiswa (
            nim,
            nama,
            email,
            password,
            id_prodi
        )
        VALUES (?, ?, ?, ?, ?)"
    );

    if (!$stmt) {
        return false;
    }

    $stmt->bind_param(
        "ssssi",
        $nim,
        $nama,
        $email,
        $password,
        $idProdi
    );

    return $stmt->execute();
}

    nama_prodi VARCHAR(100) NOT NULL UNIQUE
);

-- 3. CREATE TABLE MAHASISWA
-- DDL
-- ============================================

CREATE TABLE mahasiswa (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nim VARCHAR(20) NOT NULL UNIQUE,
    nama VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    id_prodi INT NOT NULL,
    nilai_tugas DECIMAL(5,2) DEFAULT 0,
    nilai_uts DECIMAL(5,2) DEFAULT 0,
    nilai_uas DECIMAL(5,2) DEFAULT 0,
    kehadiran INT DEFAULT 0,

    FOREIGN KEY (id_prodi)
        REFERENCES program_studi(id_prodi)
);

<?php
function ambilSemuaProdi($conn) {
    $stmt = $conn->query(
        "SELECT * FROM program_studi"
    );
    
    $execute = $stmt->execute();
    $result = $execute->get_result();
    $prodi = $result->fetch_assoc();

    return $prodi;
}

function ambilMahasiswaById($conn,$id) {
    $stmt = $conn->prepare(
        "SELECT
                mahasiswa.*,
                program_studi.nama_prodi
            FROM mahasiswa
            JOIN program_studi
                ON mahasiswa.id_prodi =
                    program_studi.id_prodi
            WHERE mahasiswa.id = ?"
    );
    $stmt->bind_param(
        "i",
        $id
    );
    $stmt->execute();

    $result =
        $stmt->get_result();
    
    return $result->fetch_assoc();
}

function updateMahasiswa(
    $conn,
    $id,
    $nim,
    $nama,
    $email,
    $idProdi
) {
    $stmt = $conn->prepare(
        "UPDATE mahasiswa
            SET
                nim = ?,
                nama = ?,
                email = ?,
                id_prodi = ?
            WHERE id = ?"
    );
    $stmt->bind_param(
        "sssii",
        $nim,
        $nama,
        $email,
        $idProdi,
        $id
    );
    return $stmt->execute();
}

function hapusMahasiswa($conn, $id) {
    $stmt = $conn->prepare(
        "DELETE
            FROM mahasiswa
            WHERE id = ?"
    );
    $stmt->bind_param(
        "i",
        $id
    );
    return $stmt->execute();
}
